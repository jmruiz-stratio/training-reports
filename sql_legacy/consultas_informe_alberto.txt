WITH user_partner AS (
    SELECT
        u.id AS userid,
        u.username,
        u.email,
        CASE
            WHEN u.email ILIKE '%stratio%' OR u.username ILIKE '%stratio%'
                THEN 'stratio'
            ELSE split_part(u.username, '-', array_length(string_to_array(u.username, '-'), 1))
        END AS partner,
        DATE_TRUNC('month', TO_TIMESTAMP(u.timecreated)) AS alta_mes
    FROM mdl_user u
    WHERE u.deleted = 0
),

-- Cohortes asignadas a usuarios
user_cohort AS (
    SELECT
        cm.userid,
        c.id   AS cohortid,
        c.name AS cohort_name
    FROM mdl_cohort_members cm
    JOIN mdl_cohort c ON c.id = cm.cohortid
),

-- Cursos a los que da acceso cada cohorte (enrolamiento por cohort)
cohort_course AS (
    SELECT
        e.courseid,
        e.customint1 AS cohortid
    FROM mdl_enrol e
    WHERE e.enrol = 'cohort'
),

-- Sesiones de dedicación con curso
dedicacion AS (
    SELECT
        s.userid,
        r.course AS courseid,
        DATE_TRUNC('month', TO_TIMESTAMP(s.login)) AS mes,
        SUM(s.duration)/3600.0 AS horas
    FROM mdl_attendanceregister_session s
    JOIN mdl_attendanceregister r ON r.id = s.register
    GROUP BY s.userid, r.course, DATE_TRUNC('month', TO_TIMESTAMP(s.login))
),

-- Resumen mensual cruzando usuario + cohort + curso
resumen AS (
    SELECT
        up.partner,
        uc.cohort_name,
        d.mes,
        COUNT(DISTINCT CASE WHEN up.alta_mes = d.mes THEN up.userid END) AS num_altas,
        SUM(d.horas) AS horas_dedicadas,
        COUNT(DISTINCT d.userid) AS usuarios_unicos
    FROM dedicacion d
    JOIN user_partner up ON up.userid = d.userid
    JOIN cohort_course cc ON cc.courseid = d.courseid
    JOIN user_cohort uc ON uc.userid = d.userid AND uc.cohortid = cc.cohortid
    GROUP BY up.partner, uc.cohort_name, d.mes
)

SELECT
    mes,
    partner,
    cohort_name,
    num_altas,
    horas_dedicadas,
    usuarios_unicos,
    SUM(num_altas)       OVER (PARTITION BY partner, cohort_name ORDER BY mes) AS acumulado_altas,
    SUM(horas_dedicadas) OVER (PARTITION BY partner, cohort_name ORDER BY mes) AS acumulado_horas,
    SUM(usuarios_unicos) OVER (PARTITION BY partner, cohort_name ORDER BY mes) AS acumulado_usuarios
FROM resumen
ORDER BY partner, cohort_name, mes;

-- Activar extensión crosstab
CREATE EXTENSION IF NOT EXISTS tablefunc;

DO $$
DECLARE
    columnas TEXT;
    sql TEXT;
BEGIN
    -- Generamos dinámicamente todas las columnas de la forma cohorte_mes
    SELECT string_agg(DISTINCT 
                      quote_ident(c.name || '_' || to_char(date_trunc('month', to_timestamp(cm.timeadded)), 'YYYY_MM')), 
                      ', ')
    INTO columnas
    FROM mdl_cohort_members cm
    JOIN mdl_cohort c ON c.id = cm.cohortid;

    -- Construimos la query dinámica pivot
    sql := format($f$
        SELECT *
        FROM crosstab(
            $$
            SELECT 
                CASE 
                    WHEN u.email ILIKE '%%stratio%%' THEN 'stratio'
                    ELSE split_part(u.username, '-', array_length(string_to_array(u.username, '-'), 1))
                END AS partner,
                c.name || '_' || TO_CHAR(date_trunc('month', TO_TIMESTAMP(cm.timeadded)), 'YYYY_MM') AS cohorte_mes,
                COUNT(*) AS num_apuntados
            FROM mdl_user u
            JOIN mdl_cohort_members cm ON cm.userid = u.id
            JOIN mdl_cohort c ON c.id = cm.cohortid
            WHERE u.deleted = 0
              AND u.suspended = 0
            GROUP BY partner, c.name, date_trunc('month', TO_TIMESTAMP(cm.timeadded))
            ORDER BY partner, cohorte_mes
            $$,
            $$
            SELECT DISTINCT c.name || '_' || TO_CHAR(date_trunc('month', TO_TIMESTAMP(cm.timeadded)), 'YYYY_MM')
            FROM mdl_cohort_members cm
            JOIN mdl_cohort c ON c.id = cm.cohortid
            ORDER BY 1
            $$
        ) AS pivot_table(partner TEXT, %s)
    $f$, columnas);

    -- Ejecutamos la query dinámica
    EXECUTE sql;
END
$$;

  SELECT DISTINCT
    up.userid,
    up.username,

    e.courseid

  WHERE (coh.name ILIKE '%14.1%' OR coh.name ILIKE '%14.6%' OR coh.name ILIKE '%14.8%')

      coh.name ILIKE '%Governance%'

      OR coh.name ILIKE '%Operaciones%'
    )
)

SELECT
  b.userid,
  b.username,
  b.email,
  b.partner,
  b.cohort_name,
  CASE
    WHEN b.cohort_name ILIKE '%Governance%'      THEN 'Governance'
    WHEN b.cohort_name ILIKE '%Data Processing%' THEN 'Data Processing'
    WHEN b.cohort_name ILIKE '%Intelligence%'    THEN 'Intelligence'
    WHEN b.cohort_name ILIKE '%Operaciones%'     THEN 'Operaciones'
    ELSE 'Otros'
  END AS cohort_group,

  c.id   AS courseid,
  c.fullname AS course_name,

  cn.finalgrade,
  TO_TIMESTAMP(cn.timemodified) AS fecha_nota,

  CASE
    WHEN cn.finalgrade IS NULL THEN 'Sin nota'
    WHEN cn.finalgrade >= 60   THEN 'Aprobado'
    ELSE 'No aprobado'
  END AS estado_aprobacion

FROM base_alumnos b
JOIN mdl_course c ON c.id = b.courseid
LEFT JOIN curso_notas cn
  ON cn.userid = b.userid
 AND cn.courseid = b.courseid

    cc.courseid
  FROM enrolled_users eu
  JOIN cohort_courses cc ON cc.cohortid = eu.cohortid
),
curso_aprobados AS (
  SELECT
    gg.userid,
    gi.courseid,
    gg.finalgrade,
    gg.timemodified AS approved_time_unix
  FROM mdl_grade_items gi
  JOIN mdl_grade_grades gg ON gg.itemid = gi.id
  WHERE gi.itemtype = 'course'
    AND gg.finalgrade IS NOT NULL
    AND gg.finalgrade >= 60
),
matricula_curso AS (
  SELECT
    ue.userid,
    e.courseid,
    MIN(ue.timecreated) AS enrol_time_unix
  FROM mdl_user_enrolments ue
  JOIN mdl_enrol e ON e.id = ue.enrolid
  GROUP BY ue.userid, e.courseid
)
SELECT
  up.partner,
  uce.userid,
  uce.cohort_name,
  uce.cohort_group,
  uce.courseid,
  c.fullname AS course_fullname,

  CASE WHEN ca.userid IS NOT NULL THEN 1 ELSE 0 END AS aprobado,
  ca.finalgrade,
  CASE
    WHEN ca.userid IS NOT NULL THEN to_timestamp(ca.approved_time_unix)
    ELSE NULL
  END AS fecha_aprobado,

  to_timestamp(
    GREATEST(
      up.user_created_unix,
      COALESCE(mc.enrol_time_unix, 0)
    )
  ) AS fecha_alta_o_matricula_mas_actual

FROM user_course_enrolments uce
JOIN user_partner up ON up.userid = uce.userid
JOIN mdl_course c ON c.id = uce.courseid
LEFT JOIN curso_aprobados ca
  ON ca.userid = uce.userid
 AND ca.courseid = uce.courseid
LEFT JOIN matricula_curso mc
  ON mc.userid = uce.userid
 AND mc.courseid = uce.courseid
ORDER BY
  up.partner,
  uce.cohort_group,
  uce.cohort_name,
  uce.userid,
  c.fullname;
